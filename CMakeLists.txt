cmake_minimum_required(VERSION 3.22.1)
project(bcrypt-tcl VERSION 2.0.1 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(BUILD_SHARED_LIBS "Build a loadable Tcl extension" ON)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
find_package(TclStub 8.6 REQUIRED)
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)

add_library(bcrypt-tcl src/library.c src/crypt/bcrypt.c src/crypt/blowfish.c)
set_target_properties(bcrypt-tcl PROPERTIES POSITION_INDEPENDENT_CODE ON
    INSTALL_RPATH_USE_LINK_PATH ON PREFIX "lib")
target_include_directories(bcrypt-tcl PRIVATE ${TCL_INCLUDE_PATH})
target_compile_definitions(bcrypt-tcl PRIVATE TCL_THREADS BUILD_bcrypt
    PROJECT_VERSION="${PROJECT_VERSION}")
target_link_libraries(bcrypt-tcl PRIVATE OpenSSL::Crypto)
if(BUILD_SHARED_LIBS)
    target_compile_definitions(bcrypt-tcl PRIVATE USE_TCL_STUBS)
    target_link_libraries(bcrypt-tcl PRIVATE ${TCL_STUB_LIBRARY})
else()
    target_compile_definitions(bcrypt-tcl PRIVATE STATIC_BUILD)
endif()

set(PACKAGE_DIR "lib/bcrypt-tcl${PROJECT_VERSION}")
install(TARGETS bcrypt-tcl LIBRARY DESTINATION ${PACKAGE_DIR}
    ARCHIVE DESTINATION ${PACKAGE_DIR} RUNTIME DESTINATION ${PACKAGE_DIR})
install(FILES src/library.h DESTINATION include/bcrypt-tcl)
install(FILES LICENSE readme.md DESTINATION ${PACKAGE_DIR})
if(BUILD_SHARED_LIBS)
    configure_file(pkgIndex.tcl.in pkgIndex.tcl @ONLY)
    install(FILES ${CMAKE_CURRENT_BINARY_DIR}/pkgIndex.tcl DESTINATION ${PACKAGE_DIR})
endif()

include(CTest)
if(BUILD_TESTING)
    if(BUILD_SHARED_LIBS)
        if(NOT TCL_TCLSH)
            message(FATAL_ERROR "Set TCL_TCLSH to run tests or disable BUILD_TESTING")
        endif()
        add_test(NAME bcrypt COMMAND ${TCL_TCLSH} ${CMAKE_CURRENT_SOURCE_DIR}/tests/all.tcl $<TARGET_FILE:bcrypt-tcl>)
    else()
        find_library(TCL_LIBRARY NAMES tcl${TCL_ABI_VERSION}
            HINTS ${TCL_LIBRARY_DIR} ${TCL_INCLUDE_PATH_PARENT}/lib REQUIRED)
        add_executable(bcrypt-static-test tests/static.c)
        target_include_directories(bcrypt-static-test PRIVATE ${TCL_INCLUDE_PATH} src)
        target_link_libraries(bcrypt-static-test PRIVATE bcrypt-tcl ${TCL_LIBRARY})
        add_test(NAME bcrypt-static COMMAND bcrypt-static-test)
    endif()
endif()
